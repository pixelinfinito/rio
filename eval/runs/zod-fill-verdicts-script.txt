def norm(s):
 return re.sub(r"(['\"])/[^'\"\n]+\1",lambda m:m.group(1)+'<absolute>/'+pathlib.PurePosixPath(m.group(0)[1:-1]).name+m.group(1),' '.join(s.split()))

 return found
